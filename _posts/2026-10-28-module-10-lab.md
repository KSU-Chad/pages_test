**Step 5 — New file: the sensors (`sensors.xacro`)**
 
This tells Gazebo that the robot's `lidar_frame` link is a lidar and its `depth_cam` link is a depth camera. The camera is set to 320 x 240 at 10 Hz so the lab computers can keep up.
 
```bash
cat > ~/ros2_ws/src/mentorpi/mentorpi_bringup/urdf/sensors.xacro <<'EOF'
<?xml version="1.0"?>
<robot xmlns:xacro="http://www.ros.org/wiki/xacro">
 
  <!-- Camera settings: lower these if the sim runs slowly -->
  <xacro:property name="cam_width" value="320"/>
  <xacro:property name="cam_height" value="240"/>
  <xacro:property name="cam_rate" value="10"/>
 
  <!-- Camera optical frame. Sim only: on the real robot the ascamera driver
       and its static transforms publish this frame. -->
  <link name="ascamera_color_0"/>
  <joint name="ascamera_color_0_joint" type="fixed">
    <parent link="depth_cam"/>
    <child link="ascamera_color_0"/>
    <origin xyz="0 0 0" rpy="${-pi/2} 0 ${-pi/2}"/>
  </joint>
 
  <!-- 2D lidar on the lidar_frame link -->
  <gazebo reference="lidar_frame">
    <sensor name="lidar" type="gpu_lidar">
      <topic>scan_raw</topic>
      <gz_frame_id>lidar_frame</gz_frame_id>
      <update_rate>10</update_rate>
      <always_on>true</always_on>
      <visualize>false</visualize>
      <lidar>
        <scan>
          <horizontal>
            <samples>450</samples>
            <resolution>1</resolution>
            <min_angle>-3.14159</min_angle>
            <max_angle>3.14159</max_angle>
          </horizontal>
        </scan>
        <!-- min 0.12 keeps the rays from hitting the robot's own parts -->
        <range>
          <min>0.12</min>
          <max>12.0</max>
          <resolution>0.01</resolution>
        </range>
        <noise>
          <type>gaussian</type>
          <mean>0.0</mean>
          <stddev>0.01</stddev>
        </noise>
      </lidar>
    </sensor>
  </gazebo>
 
  <!-- Color + depth + point cloud camera on the depth_cam link -->
  <gazebo reference="depth_cam">
    <sensor name="depth_cam" type="rgbd_camera">
      <topic>camera</topic>
      <gz_frame_id>ascamera_color_0</gz_frame_id>
      <update_rate>${cam_rate}</update_rate>
      <always_on>true</always_on>
      <visualize>false</visualize>
      <camera>
        <horizontal_fov>1.089</horizontal_fov>
        <image>
          <width>${cam_width}</width>
          <height>${cam_height}</height>
        </image>
        <clip>
          <near>0.1</near>
          <far>8.0</far>
        </clip>
        <depth_camera>
          <clip>
            <near>0.1</near>
            <far>8.0</far>
          </clip>
        </depth_camera>
      </camera>
    </sensor>
  </gazebo>
 
</robot>
EOF
```
 
**Step 6 — Replace the bridge file (`gz_bridge.yaml`)**
 
The bridge copies topics between Gazebo and ROS 2. This version adds the new lidar and camera entries. It **replaces the whole file.**
 
```bash
cat > ~/ros2_ws/src/mentorpi/mentorpi_bringup/config/gz_bridge.yaml <<'EOF'
- ros_topic_name: "clock"
  gz_topic_name: "clock"
  ros_type_name: "rosgraph_msgs/msg/Clock"
  gz_type_name: "gz.msgs.Clock"
  direction: GZ_TO_ROS
 
- ros_topic_name: "cmd_vel"
  gz_topic_name: "cmd_vel"
  ros_type_name: "geometry_msgs/msg/TwistStamped"
  gz_type_name: "gz.msgs.Twist"
  direction: ROS_TO_GZ
 
- ros_topic_name: "odom"
  gz_topic_name: "odom"
  ros_type_name: "nav_msgs/msg/Odometry"
  gz_type_name: "gz.msgs.Odometry"
  direction: GZ_TO_ROS
 
- ros_topic_name: "tf"
  gz_topic_name: "tf"
  ros_type_name: "tf2_msgs/msg/TFMessage"
  gz_type_name: "gz.msgs.Pose_V"
  direction: GZ_TO_ROS
 
- ros_topic_name: "joint_states"
  gz_topic_name: "joint_states"
  ros_type_name: "sensor_msgs/msg/JointState"
  gz_type_name: "gz.msgs.Model"
  direction: GZ_TO_ROS
 
- ros_topic_name: "scan_raw"
  gz_topic_name: "scan_raw"
  ros_type_name: "sensor_msgs/msg/LaserScan"
  gz_type_name: "gz.msgs.LaserScan"
  direction: GZ_TO_ROS
 
- ros_topic_name: "ascamera/camera_publisher/rgb0/image"
  gz_topic_name: "camera/image"
  ros_type_name: "sensor_msgs/msg/Image"
  gz_type_name: "gz.msgs.Image"
  direction: GZ_TO_ROS
 
- ros_topic_name: "ascamera/camera_publisher/rgb0/camera_info"
  gz_topic_name: "camera/camera_info"
  ros_type_name: "sensor_msgs/msg/CameraInfo"
  gz_type_name: "gz.msgs.CameraInfo"
  direction: GZ_TO_ROS
 
- ros_topic_name: "ascamera/camera_publisher/depth0/image_raw"
  gz_topic_name: "camera/depth_image"
  ros_type_name: "sensor_msgs/msg/Image"
  gz_type_name: "gz.msgs.Image"
  direction: GZ_TO_ROS
 
- ros_topic_name: "ascamera/camera_publisher/depth0/points"
  gz_topic_name: "camera/points"
  ros_type_name: "sensor_msgs/msg/PointCloud2"
  gz_type_name: "gz.msgs.PointCloudPacked"
  direction: GZ_TO_ROS
EOF
```

## Troubleshooting
 
| Symptom | Likely cause | Fix |
|---|---|---|
| Gazebo opens but no walls | World didn't load | Check the launch command and the world path |
| `gz topic -l` has no sensor topics | Sensors plugin missing in the world | Check `my_world.sdf` still has the `gz-sim-sensors-system` plugin |
| `ros2 topic list` has no `scan_raw` | Bridge died | Read the Terminal 1 output for `process has died` |
| LaserScan blank in RViz | QoS or Fixed Frame | Set Best Effort, set Fixed Frame to `odom` |
| Everything is slow (rates low) | Lab machine rendering | Measure with `--use-sim-time`. Close other windows |
| Robot doesn't move | Gate not running or motion disabled | Start the gate, check `/enable_motion` |
| RViz shows "no transform" | RViz not on sim time | Restart with `use_sim_time:=true` |
| Point cloud rotated | Known frame issue | Record what you see (Part 7) |
